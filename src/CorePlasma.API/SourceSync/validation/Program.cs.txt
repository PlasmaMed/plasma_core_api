using System.Text.Json;
using CorePlasma.API.SourceSync;
using CorePlasma.Common.Exceptions;

if (Environment.GetEnvironmentVariable("SOURCE_SYNC_COMPARE_REPORT") is { Length: > 0 } compareReport)
{
    await CompareCases.RunAsync(compareReport);
    return; // Read-only comparison mode never runs fixture installation or DML.
}

var checks = 0;
void Check(bool valid, string name) { if (!valid) throw new Exception(name); checks++; }
void Reject(Action action, string name)
{
    try { action(); } catch (BusinessException) { checks++; return; }
    throw new Exception("Expected rejection: " + name);
}
JsonElement Parse(string value) => JsonDocument.Parse(value).RootElement.Clone();
string Hex(params string?[] values) => Convert.ToHexString(SourceSyncCodec.Hash(values));
var table = new SourceSyncTable("Synthetic", ["id", "value", "nullable"], ["id"], "catalog-v1");
SourceSyncCodec.Tables([table]);
Check(Hex((string?)null) != Hex(""), "Null differs from empty string");
Check(Hex("ab", "c") != Hex("a", "bc"), "Length prefix prevents field boundary collision");
Check(Hex("é") != Hex("e\u0301"), "No Unicode normalization");
Check(Hex("01.000") != Hex("1"), "Numeric lexical values remain distinct");
Check(Hex(" {\"n\": 1} ") != Hex("{\"n\":1}"), "JSON lexical values remain distinct");
Check(Hex("a\0b") != Hex("ab"), "Embedded null preserved");
Check(SourceSyncCodec.DecimalId("9223372036854775807") == long.MaxValue, "Large decimal identity is exact");
Reject(() => SourceSyncCodec.DecimalId("01"), "Noncanonical decimal");
Reject(() => SourceSyncCodec.DecimalId("9223372036854775808"), "Int64 overflow");
Reject(() => SourceSyncCodec.DecimalId("1e3"), "Numeric exponent ID");
Reject(() => SourceSyncCodec.Row(Parse("{\"id\":1,\"value\":\"x\",\"nullable\":null}"), table), "JSON numbers rejected");
Reject(() => SourceSyncCodec.Row(Parse("{\"id\":\"1\",\"id\":\"2\",\"value\":\"x\",\"nullable\":null}"), table), "Duplicate fields rejected");
Reject(() => SourceSyncCodec.Row(Parse("{\"id\":\"1\",\"value\":\"x\"}"), table), "Missing field rejected");
Reject(() => SourceSyncCodec.Row(Parse("{\"id\":\"1\",\"value\":\"x\",\"nullable\":null,\"extra\":null}"), table), "Unknown field rejected");
Reject(() => SourceSyncCodec.Row(Parse("{\"id\":null,\"value\":\"x\",\"nullable\":null}"), table), "Null PK rejected");
var rowA = Parse("{\"id\":\"1\",\"value\":\"x\",\"nullable\":null}");
var rowB = Parse("{\"nullable\":null,\"value\":\"x\",\"id\":\"1\"}");
Check(SourceSyncCodec.Row(rowA, table).SequenceEqual(SourceSyncCodec.Row(rowB, table)), "Object property order does not change ordered vector");
var inserted = new SourceSyncEvent("1", "1", "Synthetic", "INSERT", Parse("null"), rowA, "2026-11-09T00:00:00Z");
Check(SourceSyncCodec.EventHash(inserted, table).SequenceEqual(SourceSyncCodec.EventHash(inserted with { NewRow = rowB }, table)), "Equivalent reordered event repeats immutable hash");
Check(!SourceSyncCodec.EventHash(inserted, table).SequenceEqual(SourceSyncCodec.EventHash(inserted with { TransactionId = "2" }, table)), "Transaction identity contributes to immutable hash");
Reject(() => SourceSyncCodec.EventHash(inserted with { OccurredAt = "2026-11-09T07:00:00+07:00" }, table), "Non UTC timestamp");
Reject(() => SourceSyncCodec.EventHash(inserted with { OldRow = rowA }, table), "Insert before-image must be null");
Reject(() => SourceSyncCodec.EventHash(inserted with { Operation = "TRUNCATE" }, table), "Truncate after-image must be null");
Reject(() => SourceSyncCodec.Tables([table with { Columns = ["id", "ID"] }]), "SQL case collision rejected");
Reject(() => SourceSyncCodec.Tables([table with { Columns = ["__ss_key_hash"] }]), "Reserved hash column rejected");
Reject(() => SourceSyncCodec.Schema("dbo"), "Unapproved namespace rejected");
Check(SourceSyncCodec.Quote("odd]name") == "[odd]]name]", "Identifier quote is safe");
try
{
    JsonSerializer.Deserialize<SourceSyncBegin>("{\"installationId\":\"11111111-1111-1111-1111-111111111111\",\"generation\":\"22222222-2222-2222-2222-222222222222\",\"snapshot\":\"1:2:\",\"tables\":[],\"unknown\":true}", SourceSyncCodec.Json);
    throw new Exception("Unknown envelope fields must fail.");
}
catch (JsonException) { checks++; }
Console.WriteLine($"{checks} source-sync contract checks passed.");
await SqlCases.RunAsync();
